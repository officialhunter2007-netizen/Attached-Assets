// ─── STATE ───────────────────────────────────────────────────────────────────
const answers = {};
let submitted = false;

// ─── MCQ ─────────────────────────────────────────────────────────────────────
document.querySelectorAll('.option input[type="radio"]').forEach(radio => {
  radio.addEventListener('change', function(){
    if(submitted) return;
    const card = this.closest('.q-card');
    const qid = card.dataset.qid;
    card.querySelectorAll('.option').forEach(o => o.classList.remove('selected'));
    this.closest('.option').classList.add('selected');
    answers[qid] = this.value;
    updateProgress();
  });
});

// ─── FILL ─────────────────────────────────────────────────────────────────────
document.querySelectorAll('.fill-input').forEach(inp => {
  inp.addEventListener('input', function(){
    if(submitted) return;
    const card = this.closest('.q-card');
    answers[card.dataset.qid] = this.value.trim().toUpperCase();
    updateProgress();
  });
});

// ─── TRUE/FALSE ───────────────────────────────────────────────────────────────
function selectTF(btn, qid) {
  if(submitted) return;
  const card = btn.closest('.q-card');
  card.querySelectorAll('.tf-btn').forEach(b => b.classList.remove('selected'));
  btn.classList.add('selected');
  answers[qid] = btn.dataset.val;
  updateProgress();
}

// ─── MATCH ───────────────────────────────────────────────────────────────────
['m4a','m4b','m4c','m4d'].forEach(id => {
  const el = document.getElementById(id);
  if(el) {
    el.addEventListener('change', () => {
      if(submitted) return;
      answers[4] = {
        a: document.getElementById('m4a').value,
        b: document.getElementById('m4b').value,
        c: document.getElementById('m4c').value,
        d: document.getElementById('m4d').value
      };
      updateProgress();
    });
  }
});

// ─── DRAG SORT ────────────────────────────────────────────────────────────────
let dragSrc = null;
document.querySelectorAll('.sort-list').forEach(list => {
  list.querySelectorAll('.sort-item').forEach(item => {
    item.addEventListener('dragstart', function(e){
      dragSrc = this;
      this.classList.add('dragging');
      e.dataTransfer.effectAllowed = 'move';
    });
    item.addEventListener('dragend', function(){
      this.classList.remove('dragging');
    });
    item.addEventListener('dragover', function(e){
      e.preventDefault();
      if(dragSrc && dragSrc !== this) {
        const list = this.parentNode;
        const items = [...list.querySelectorAll('.sort-item')];
        const srcIdx = items.indexOf(dragSrc);
        const tgtIdx = items.indexOf(this);
        if(srcIdx < tgtIdx) list.insertBefore(dragSrc, this.nextSibling);
        else list.insertBefore(dragSrc, this);
      }
    });
    item.addEventListener('drop', function(e){ e.preventDefault(); });
  });
});

// ─── PROGRESS ────────────────────────────────────────────────────────────────
function countAnswered() {
  let n = 0;
  for(let qid=1; qid<=10; qid++) {
    const v = answers[qid];
    if(v !== undefined && v !== null && v !== '') {
      if(typeof v === 'object') {
        const vals = Object.values(v);
        if(vals.some(x => x !== '')) n++;
      } else n++;
    }
  }
  return n;
}
function updateProgress() {
  const n = countAnswered();
  const pct = Math.round((n/10)*100);
  document.getElementById('progressFill').style.width = pct+'%';
  document.getElementById('progressLabel').textContent = n+' / 10';
}

function checkMatch4() {
  const a = answers[4];
  if(!a) return false;
  return a.a === 'and' && a.b === 'or' && a.c === 'xor' && a.d === 'not';
}

function checkSort6() {
  const items = [...document.querySelectorAll('#sort6 .sort-item')];
  const userOrder = items.map(i => i.dataset.order);
  return JSON.stringify(userOrder) === JSON.stringify(['1','2','3','4']);
}

function submitQuiz() {
  submitted = true;
  document.getElementById('submitBtn').disabled = true;

  let totalScore = 0;
  const sections = [];

  // Grade MCQ / TF / FILL
  for(const [qid, cfg] of Object.entries(CORRECT)) {
    const card = document.querySelector(`.q-card[data-qid="${qid}"]`);
    if(!card) continue;
    const fb = card.querySelector('.q-feedback');
    let correct = false;

    if(card.dataset.type === 'fill') {
      const inp = card.querySelector('.fill-input');
      const userVal = (inp.value.trim().toUpperCase() || '').replace(/\s/g,'');
      correct = userVal === cfg.ans;
      inp.classList.add(correct ? 'correct-ans' : 'wrong-ans');
    } else if(card.dataset.type === 'mcq') {
      const userVal = answers[qid];
      correct = userVal === cfg.ans;
      card.querySelectorAll('.option').forEach(opt => {
        const radio = opt.querySelector('input');
        if(radio.value === cfg.ans) opt.classList.add('correct-ans');
        else if(radio.value === userVal && !correct) opt.classList.add('wrong-ans');
      });
    } else if(card.dataset.type === 'tf') {
      const userVal = answers[qid];
      correct = userVal === cfg.ans;
      card.querySelectorAll('.tf-btn').forEach(btn => {
        if(btn.dataset.val === cfg.ans) btn.classList.add('correct-ans');
        else if(btn.dataset.val === userVal && !correct) btn.classList.add('wrong-ans');
      });
    }

    card.classList.add(correct ? 'correct' : 'wrong');
    fb.textContent = correct ? '✓ ' + cfg.fb_ok : '✗ ' + cfg.fb_err;
    fb.className = 'q-feedback show ' + (correct ? 'ok' : 'err');
    if(correct) totalScore += 10;
    sections.push({ qid, correct, pts: correct ? 10 : 0 });
  }

  // Match Q4
  {
    const card = document.querySelector('.q-card[data-qid="4"]');
    const correct = checkMatch4();
    const fb = card.querySelector('.q-feedback');
    card.classList.add(correct ? 'correct' : 'wrong');
    fb.textContent = correct ? '✓ طابقت جميع البوابات بشكل صحيح!' : '✗ الإجابة: AND (كلها 1)، OR (أحدها 1)، XOR (مختلفة)، NOT (عكس).';
    fb.className = 'q-feedback show ' + (correct ? 'ok' : 'err');
    if(correct) totalScore += 10;
    sections.push({ qid: 4, correct, pts: correct ? 10 : 0 });
  }

  // Sort Q6
  {
    const card = document.querySelector('.q-card[data-qid="6"]');
    const correct = checkSort6();
    const fb = card.querySelector('.q-feedback');
    card.classList.add(correct ? 'correct' : 'wrong');
    fb.textContent = correct ? '✓ الترتيب صحيح: جدول الحقيقة → رسم الخريطة → التجميع → التعبير.' : '✗ الترتيب: (1) جدول الحقيقة (2) رسم الخريطة (3) التجميع (4) التعبير.';
    fb.className = 'q-feedback show ' + (correct ? 'ok' : 'err');
    if(correct) totalScore += 10;
    sections.push({ qid: 6, correct, pts: correct ? 10 : 0 });
  }

  showResult(totalScore, sections);

  // =====================================================================
  // القاعدة الأساسية: إرسال النتيجة إلى النظام باستخدام الدالة المطلوبة
  // =====================================================================
  if (typeof window.submitScore === 'function') {
    window.submitScore(totalScore);
  } else {
    console.log("تم حساب النتيجة: " + totalScore + " (دالة window.submitScore سيتم استدعاؤها في بيئتك)");
  }
}

// ─── RESULT DISPLAY ───────────────────────────────────────────────────────────
function showResult(score, sections) {
  document.getElementById('quizContainer').style.display = 'none';
  const rs = document.getElementById('resultScreen');
  rs.classList.add('show');

  const circumference = 314;
  const offset = circumference - (score / 100) * circumference;
  const ring = document.getElementById('ringVal');
  const color = score >= 80 ? '#22c55e' : score >= 60 ? '#f59e0b' : '#ef4444';
  ring.style.stroke = color;
  setTimeout(() => { ring.style.strokeDashoffset = offset; }, 100);

  let n = 0;
  const numEl = document.getElementById('scoreNum');
  numEl.style.color = color;
  const timer = setInterval(() => {
    n = Math.min(n + 2, score);
    numEl.textContent = n;
    if(n >= score) clearInterval(timer);
  }, 20);

  const gradeEl = document.getElementById('resultGrade');
  const subEl   = document.getElementById('resultSub');
  if(score >= 90) { gradeEl.textContent = '🏆 ممتاز'; gradeEl.style.color = '#22c55e'; subEl.textContent = 'أداء استثنائي وفهم عميق للمنطق الرقمي.'; }
  else if(score >= 70) { gradeEl.textContent = '⭐ جيد جداً'; gradeEl.style.color = '#4f8ef7'; subEl.textContent = 'فهم متين للأساسيات مع بعض الأخطاء البسيطة.'; }
  else if(score >= 50) { gradeEl.textContent = '✔ مقبول'; gradeEl.style.color = '#f59e0b'; subEl.textContent = 'اجتزت الاختبار، لكن يُنصح بمراجعة الأخطاء.'; }
  else { gradeEl.textContent = '📚 يحتاج مراجعة'; gradeEl.style.color = '#ef4444'; subEl.textContent = 'ننصح بمراجعة وحدة المنطق الرقمي وأنظمة الترقيم.'; }

  const sectionNames = {
    1:'التحويل الثنائي للعشري', 2:'التحويل للسادس عشري', 3:'قانون دي مورغان',
    4:'البوابات المنطقية', 5:'خرائط كارنو', 6:'خطوات K-map',
    7:'الدوائر التسلسلية', 8:'Hex إلى IPv4', 9:'الجمع الثنائي', 10:'استخدام Hex'
  };

  const bd = document.getElementById('breakdown');
  bd.innerHTML = '<div style="font-size:13px;font-weight:600;color:var(--muted);margin-bottom:8px;padding-bottom:8px;border-bottom:1px solid var(--border)">تفصيل الدرجات</div>';
  sections.sort((a,b) => a.qid - b.qid).forEach(s => {
    bd.innerHTML += `
      <div class="breakdown-row">
        <span>س${s.qid}: ${sectionNames[s.qid] || ''}</span>
        <span class="pts ${s.correct?'ok':'err'}">${s.correct ? '+10 ✓' : '0 ✗'}</span>
      </div>`;
  });
  bd.innerHTML += `<div class="breakdown-row" style="font-weight:700;font-size:15px;margin-top:4px"><span>المجموع</span><span class="pts" style="color:${color}">${score} / 100</span></div>`;

  window.scrollTo({ top: 0, behavior: 'smooth' });
}
