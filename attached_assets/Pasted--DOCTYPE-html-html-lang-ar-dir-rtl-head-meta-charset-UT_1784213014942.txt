        let audioCtx = null;

        function playSound(type) {
            try {
                if (!audioCtx) audioCtx = new(window.AudioContext || window.webkitAudioContext)();
                if (audioCtx.state === 'suspended') audioCtx.resume();
                const osc = audioCtx.createOscillator();
                const gain = audioCtx.createGain();
                osc.connect(gain);
                gain.connect(audioCtx.destination);
                const now = audioCtx.currentTime;

                if (type === 'click') {
                    osc.type = 'sine';
                    osc.frequency.setValueAtTime(600, now);
                    gain.gain.setValueAtTime(0.1, now);
                    gain.gain.exponentialRampToValueAtTime(0.01, now + 0.1);
                    osc.start(now);
                    osc.stop(now + 0.1);
                } else if (type === 'on') {
                    osc.type = 'triangle';
                    osc.frequency.setValueAtTime(400, now);
                    osc.frequency.setValueAtTime(800, now + 0.1);
                    gain.gain.setValueAtTime(0.1, now);
                    gain.gain.linearRampToValueAtTime(0.01, now + 0.3);
                    osc.start(now);
                    osc.stop(now + 0.3);
                } else if (type === 'off') {
                    osc.type = 'sawtooth';
                    osc.frequency.setValueAtTime(300, now);
                    osc.frequency.linearRampToValueAtTime(150, now + 0.2);
                    gain.gain.setValueAtTime(0.1, now);
                    gain.gain.linearRampToValueAtTime(0.01, now + 0.2);
                    osc.start(now);
                    osc.stop(now + 0.2);
                }
            } catch (e) {}
        }

        function setInput(id, val) {
            const el = document.getElementById(id);
            el.innerHTML = `<bdi dir="ltr">${val}</bdi>`;
            if (val === 1) {
                el.className = 'input-node input-1';
            } else {
                el.className = 'input-node input-0';
            }
        }

        function setOutput(val) {
            const bulb = document.getElementById('output-bulb');
            const outVal = document.getElementById('output-val');
            const wireOut = document.getElementById('wire-out');

            outVal.innerHTML = `<bdi dir="ltr">${val}</bdi>`;

            if (val === 1) {
                bulb.className = 'fas fa-lightbulb bulb bulb-on';
                outVal.className = 'math-text text-xl font-bold text-yellow-400 mt-2';
                wireOut.classList.add('active');
                playSound('on');
            } else {
                bulb.className = 'fas fa-lightbulb bulb bulb-off';
                outVal.className = 'math-text text-xl font-bold text-slate-500 mt-2';
                wireOut.classList.remove('active');
                playSound('off');
            }
        }

        function setGate(name, colorClass) {
            const gate = document.getElementById('gate-box');
            gate.innerHTML = name;
            gate.className = `gate-box ${colorClass} math-text`;
            document.getElementById('gate-title').innerHTML = `بوابة ${name}`;
        }

        function setEquation(eq) {
            const eqEl = document.getElementById('math-equation');
            eqEl.innerHTML = `<bdi dir="ltr">${eq}</bdi>`;
            eqEl.style.opacity = '1';
        }

        let currentStep = 0;

            currentStep = 0;
            document.getElementById('explanation').innerHTML =
                "مرحباً بك! الكمبيوتر يفكر باستخدام دوائر بسيطة جداً تسمى <b>البوابات المنطقية</b>. هذه البوابات تستقبل إشارات (0 أو 1) وتُخرج نتيجة بناءً على شرط معين. اضغط على <b>'الخطوة التالية'</b> لنبدأ بأول وأشهر بوابة.";
            document.getElementById('visual-area').classList.add('opacity-30');
            document.getElementById('btn-next').style.display = '';

            document.getElementById('wire-out').classList.remove('active');

            setInput('input-b', 0);

            bulb.className = 'fas fa-lightbulb bulb bulb-off';
            const outVal = document.getElementById('output-val');
            outVal.innerHTML = '<bdi dir="ltr">0</bdi>';
            outVal.className = 'math-text text-xl font-bold text-slate-500 mt-2';
            playSound('click');
        }

        function nextStep() {
            if (document.getElementById('btn-next').style.display === 'none') return;
            playSound('click');
            if (currentStep < steps.length) {
                const stepData = steps[currentStep];
                document.getElementById('explanation').innerHTML = stepData.text;
                if (stepData.action) {
                    stepData.action();
                }
                currentStep++;
            }
